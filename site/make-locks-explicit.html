/** thread safe */
class Counter {

  /** guarded by 'this' */
  private long count;

  synchronized void increment(long increment) {
    if (increment < 1) {
      throw new IllegalArgumentException();
    }

    var next = count;
    next = next + increment;
    count = next;
  }
}

/** thread safe */
class Counter {

  /** guarded by 'this' */
  private long count;

  void increment(long increment) {
    synchronized (this) {
      if (increment < 1) {
        throw new IllegalArgumentException();
      }

      var next = count;
      next = next + increment;
      count = next;
    }
  }
}
